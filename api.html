API Methods & Code Snippets

curl -X POST "https://ghbooster.com/api/v2" \
  -d "key=e984f8a192bc84591a2741890f892a7f" \
  -d "action=add" \
  -d "service=101" \
  -d "link=https://instagram.com/appiahattah" \
  -d "quantity=1000"
<?php
$ch = curl_init('https://ghbooster.com/api/v2');
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POSTFIELDS, [
    'key' => 'e984f8a192bc84591a2741890f892a7f',
    'action' => 'add',
    'service' => 101,
    'link' => 'https://instagram.com/appiahattah',
    'quantity' => 1000
]);
$response = curl_exec($ch);
curl_close($ch);
print_r(json_decode($response, true));
?>
import requests

url = "https://ghbooster.com/api/v2"
payload = {
    "key": "e984f8a192bc84591a2741890f892a7f",
    "action": "add",
    "service": 101,
    "link": "https://instagram.com/appiahattah",
    "quantity": 1000
}

response = requests.post(url, data=payload)
print(response.json())
const formData = new URLSearchParams();
formData.append('key', 'e984f8a192bc84591a2741890f892a7f');
formData.append('action', 'add');
formData.append('service', '101');
formData.append('link', 'https://instagram.com/appiahattah');
formData.append('quantity', '1000');

fetch('https://ghbooster.com/api/v2', {
  method: 'POST',
  body: formData
})
.then(res => res.json())
.then(data => console.log(data));

API Parameters Reference

Parameter Type Required Description
key String Yes Your secret API key from account settings.
action String Yes Action type: services, add, status, balance.
service Integer For 'add' ID of the requested service (e.g. 101, 104).
link String For 'add' Target profile or video URL.
quantity Integer For 'add' Requested order quantity.
